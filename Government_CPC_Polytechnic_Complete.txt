const courseOrder = ["cse","ai","mechanical","mtt","ece","eee","civil","auto","cp"];

/* =========================================================
   COURSE RENDERING
   ========================================================= */
function renderCourses(filter="all"){
    const grid = document.getElementById("courseGrid");
    grid.innerHTML = "";

    courseOrder.forEach(key=>{
        const data = departments[key];
        if(filter !== "all" && data.category !== filter) return;

        const card = document.createElement("article");
        card.className = "course-card reveal show";
        card.dataset.category = data.category;
        card.addEventListener("click",()=>openDepartment(key));

        const preview = data.overview.slice(0,4).map(item=>`<li>${escapeHtml(item)}</li>`).join("");

        card.innerHTML = `
            <div class="course-card-inner">
                <div class="course-front">
                    <div class="course-image">
                        <img src="${courseImages[key]}" alt="${escapeHtml(data.title)}">
                        <span class="course-code">${escapeHtml(data.code)}</span>
                    </div>
                    <div class="course-info">
                        <h3>${escapeHtml(data.title)}</h3>
                        <p>${escapeHtml(data.overview.slice(0,4).join(" · "))}</p>
                    </div>
                </div>
                <div class="course-back">
                    <h3>${escapeHtml(shortName(data.title))}</h3>
                    <p>${escapeHtml(data.description)}</p>
                    <ul>${preview}</ul>
                    <div class="course-open">OPEN DEPARTMENT</div>
                </div>
            </div>
        `;
        grid.appendChild(card);
    });
}

function shortName(title){
    const names = {
        "Computer Science & Engineering":"CSE",
        "Artificial Intelligence & Machine Learning":"AI / ML",
        "Mechanical Engineering":"Mechanical",
        "Machine Tool Technology":"MTT",
        "Electronics & Communication Engineering":"ECE",
        "Electrical & Electronics Engineering":"EEE",
        "Civil Engineering":"Civil",
        "Automobile Engineering":"Automobile",
        "Commercial Practice":"Commercial Practice"
    };
    return names[title] || title;
}

function filterCourses(category,button){
    document.querySelectorAll(".filter-btn").forEach(btn=>btn.classList.remove("active"));
    button.classList.add("active");
    renderCourses(category);
}

/* =========================================================
   DEPARTMENT MODAL
   ========================================================= */
function openDepartment(key){
    const data = departments[key];
    if(!data) return;

    document.body.classList.add("modal-open");
    document.getElementById("departmentCode").textContent = data.code;
    document.getElementById("departmentTitle").textContent = data.title;
    document.getElementById("departmentDescription").textContent =
        `${data.description} Programme: 3-Year Diploma / learning pathway.`;

    document.getElementById("departmentPhoto").src = data.image;
    document.getElementById("departmentPhoto").alt = data.title;

    fillList("overviewList",data.overview);

function openSearch(){
    document.getElementById("searchOverlay").classList.add("show");
    document.body.classList.add("modal-open");
    setTimeout(()=>document.getElementById("globalSearch").focus(),100);
    globalSearch();
}

function closeSearch(){
    document.getElementById("searchOverlay").classList.remove("show");
    document.body.classList.remove("modal-open");
}

function globalSearch(){
    const query = document.getElementById("globalSearch").value.trim().toLowerCase();
    const results = document.getElementById("searchResults");
    results.innerHTML = "";

    const matches = searchItems.filter(item=>{
        if(!query) return true;
        return `${item.title} ${item.text}`.toLowerCase().includes(query);
    }).slice(0,12);

    if(!matches.length){
        results.innerHTML = `<div class="search-result"><strong>No results</strong><span>Try another keyword.</span></div>`;
        return;
    }

    matches.forEach(item=>{
        const result = document.createElement("div");
        result.className = "search-result";
        result.innerHTML = `<strong>${escapeHtml(item.title)}</strong><span>${escapeHtml(item.text)}</span>`;
        result.addEventListener("click",()=>{
            closeSearch();
            item.action();
        });
        results.appendChild(result);
    });
}

/* =========================================================
   REVEAL ANIMATION
   ========================================================= */
const revealObserver = new IntersectionObserver(entries=>{
    entries.forEach(entry=>{
        if(entry.isIntersecting){
            entry.target.classList.add("show");
            revealObserver.unobserve(entry.target);
        }
    });
},{threshold:.12});

function observeReveals(){
    document.querySelectorAll(".reveal").forEach(element=>revealObserver.observe(element));
}

/* =========================================================
   COUNTERS
   ========================================================= */
const counterObserver = new IntersectionObserver(entries=>{
    entries.forEach(entry=>{
        if(!entry.isIntersecting) return;

        const element = entry.target;
        const target = Number(element.dataset.count);
        let current = 0;
        const duration = 1200;
        const start = performance.now();

        function tick(now){
            const progress = Math.min((now-start)/duration,1);
            current = Math.floor(progress*target);
            element.textContent = current;
            if(progress < 1){
                requestAnimationFrame(tick);
            }else{
                element.textContent = target;
            }
        }
        requestAnimationFrame(tick);
        counterObserver.unobserve(element);
    });
},{threshold:.5});

function escapeHtml(value){
    return String(value)
        .replaceAll("&","&amp;")
        .replaceAll("<","&lt;")
        .replaceAll(">","&gt;")
        .replaceAll('"',"&quot;")
        .replaceAll("'","&#039;");
}
